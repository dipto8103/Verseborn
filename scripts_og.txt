# Initialize Groq client for the coding agent
client = Groq()

def groq_generate_content(system_instruction, user_prompt):
    """
    Generate content using Groq API with system and user prompts.
    """
    messages = [
        {"role": "system", "content": system_instruction},
        {"role": "user", "content": user_prompt}
    ]
    
    completion = client.chat.completions.create(
        model="llama-3.3-70b-versatile",
        messages=messages,
        temperature=1,
        max_completion_tokens=5000,
        top_p=1,
        stream=False,
        stop=None
    )
    
    return completion.choices[0].message.content
