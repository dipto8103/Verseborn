import bpy
import math

world = bpy.context.scene.world
if world and world.use_nodes:
    nodes = world.node_tree.nodes
    links = world.node_tree.links
    
    for node in nodes:
        nodes.remove(node)

    sky_texture_node = nodes.new(type='ShaderNodeTexSky')
    sky_texture_node.location = (-200, 300)
    sky_texture_node.sky_type = 'NISHITA'
    sky_texture_node.sun_disc = True
    sky_texture_node.sun_size = math.radians(0.526)
    sky_texture_node.sun_intensity = 1.0
    sky_texture_node.sun_elevation = math.radians(25.0)
    sky_texture_node.sun_rotation = math.radians(150.0)
    sky_texture_node.altitude = 1000.0
    sky_texture_node.air_density = 1.0
    sky_texture_node.dust_density = 0.2
    sky_texture_node.ozone_density = 1.0

    world_output_node = nodes.new(type='ShaderNodeOutputWorld')
    world_output_node.location = (200, 300)

    links.new(sky_texture_node.outputs['Color'], world_output_node.inputs['Surface'])

    background_node = nodes.new(type='ShaderNodeBackground')
    background_node.location = (-200, 100)
    background_node.inputs['Strength'].default_value = 0.5
    
    volume_scatter_node = nodes.new(type='ShaderNodeVolumeScatter')
    volume_scatter_node.location = (0, 100)
    volume_scatter_node.inputs['Density'].default_value = 0.02
    volume_scatter_node.inputs['Anisotropy'].default_value = 0.2
    
    links.new(background_node.outputs['Background'], world_output_node.inputs['Volume'])
    links.new(volume_scatter_node.outputs['Volume'], background_node.inputs['Color'])
