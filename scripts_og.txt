import bpy
import random
from math import radians

def create_flying_car(location):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    car_body = bpy.context.object
    car_body.name = "FlyingCar_Body"
    car_body.scale = (4, 2, 0.8)
    bpy.ops.object.transform_apply(scale=True)

    bpy.ops.mesh.primitive_cube_add(size=1, location=(location[0] + 0.8, location[1], location[2] + 0.6))
    cockpit = bpy.context.object
    cockpit.name = "FlyingCar_Cockpit"
    cockpit.scale = (1.5, 1.8, 0.7)
    
    cockpit.parent = car_body
    
    bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=0.5, location=(location[0] - 2.2, location[1] + 0.7, location[2] - 0.2))
    thruster_l = bpy.context.object
    thruster_l.name = "FlyingCar_Thruster_L"
    thruster_l.rotation_euler[0] = radians(90)
    thruster_l.parent = car_body

    bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=0.5, location=(location[0] - 2.2, location[1] - 0.7, location[2] - 0.2))
    thruster_r = bpy.context.object
    thruster_r.name = "FlyingCar_Thruster_R"
    thruster_r.rotation_euler[0] = radians(90)
    thruster_r.parent = car_body

    body_mat = bpy.data.materials.new(name="CarBodyMaterial")
    body_mat.use_nodes = True
    principled_bsdf = body_mat.node_tree.nodes.get('Principled BSDF')
    if principled_bsdf:
        principled_bsdf.inputs['Base Color'].default_value = (0.1, 0.1, 0.1, 1)
        principled_bsdf.inputs['Metallic'].default_value = 1.0
        principled_bsdf.inputs['Roughness'].default_value = 0.2
    car_body.data.materials.append(body_mat)

    cockpit_mat = bpy.data.materials.new(name="CockpitMaterial")
    cockpit_mat.use_nodes = True
    principled_bsdf_cockpit = cockpit_mat.node_tree.nodes.get('Principled BSDF')
    if principled_bsdf_cockpit:
        principled_bsdf_cockpit.inputs['Base Color'].default_value = (0.05, 0.05, 0.15, 1)
        principled_bsdf_cockpit.inputs['Roughness'].default_value = 0.1
        principled_bsdf_cockpit.inputs['Transmission Weight'].default_value = 1.0
        principled_bsdf_cockpit.inputs['IOR'].default_value = 1.45
    cockpit.data.materials.append(cockpit_mat)

    thruster_mat = bpy.data.materials.new(name="ThrusterGlow")
    thruster_mat.use_nodes = True
    principled_bsdf_thruster = thruster_mat.node_tree.nodes.get('Principled BSDF')
    if principled_bsdf_thruster:
        principled_bsdf_thruster.inputs['Emission Color'].default_value = (0.1, 0.5, 1, 1)
        principled_bsdf_thruster.inputs['Emission Strength'].default_value = 15.0
    thruster_l.data.materials.append(thruster_mat)
    thruster_r.data.materials.append(thruster_mat)

    car_body.rotation_euler[2] = radians(random.uniform(-45, 45))
    return car_body

def create_automated_bus(location):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    bus_body = bpy.context.object
    bus_body.name = "AutomatedBus_Body"
    bus_body.scale = (12, 3, 2.5)
    
    bus_mat = bpy.data.materials.new(name="BusBodyMaterial")
    bus_mat.use_nodes = True
    bsdf = bus_mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (0.8, 0.8, 0.85, 1)
        bsdf.inputs['Metallic'].default_value = 0.8
        bsdf.inputs['Roughness'].default_value = 0.25
    bus_body.data.materials.append(bus_mat)
    
    bpy.ops.mesh.primitive_cube_add(location=(location[0], location[1], location[2]))
    window = bpy.context.object
    window.name = "Bus_Window"
    window.scale = (11, 3.05, 1.5)
    window.parent = bus_body
    
    window_mat = bpy.data.materials.new(name="BusWindowMaterial")
    window_mat.use_nodes = True
    bsdf_win = window_mat.node_tree.nodes.get('Principled BSDF')
    if bsdf_win:
        bsdf_win.inputs['Base Color'].default_value = (0.01, 0.01, 0.02, 1)
        bsdf_win.inputs['Roughness'].default_value = 0.05
        bsdf_win.inputs['Transmission Weight'].default_value = 1.0
        bsdf_win.inputs['IOR'].default_value = 1.52
    window.data.materials.append(window_mat)

    return bus_body

def create_stylized_pedestrian(location):
    
    bpy.ops.mesh.primitive_ico_sphere_add(radius=0.25, subdivisions=2, location=(location[0], location[1], location[2] + 1.5))
    head = bpy.context.object

    bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=1.2, location=(location[0], location[1], location[2] + 0.6))
    torso = bpy.context.object
    
    bpy.ops.mesh.primitive_cylinder_add(radius=0.1, depth=1.0, location=(location[0], location[1] + 0.4, location[2] + 0.7))
    arm_l = bpy.context.object
    arm_l.rotation_euler[0] = radians(15)

    bpy.ops.mesh.primitive_cylinder_add(radius=0.1, depth=1.0, location=(location[0], location[1] - 0.4, location[2] + 0.7))
    arm_r = bpy.context.object
    arm_r.rotation_euler[0] = radians(-15)

    bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=1.0, location=(location[0], location[1] + 0.15, location[2] - 0.5))
    leg_l = bpy.context.object

    bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=1.0, location=(location[0], location[1] - 0.15, location[2] - 0.5))
    leg_r = bpy.context.object
    
    ped_mat = bpy.data.materials.new(name="PedestrianMaterial")
    ped_mat.use_nodes = True
    bsdf = ped_mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (random.random(), random.random(), random.random(), 1)
        bsdf.inputs['Roughness'].default_value = 0.8
    
    for obj in [head, torso, arm_l, arm_r, leg_l, leg_r]:
        obj.data.materials.append(ped_mat)

    bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
    parent_empty = bpy.context.object
    parent_empty.name = "Pedestrian"

    for obj in [head, torso, arm_l, arm_r, leg_l, leg_r]:
        obj.parent = parent_empty
        
    return parent_empty

def create_futuristic_tree(location):
    bpy.ops.mesh.primitive_cylinder_add(vertices=8, radius=0.3, depth=5, location=(location[0], location[1], location[2] + 2.5))
    trunk = bpy.context.object
    trunk.name = "Tree_Trunk"

    trunk_mat = bpy.data.materials.new(name="TrunkMaterial")
    trunk_mat.use_nodes = True
    bsdf_trunk = trunk_mat.node_tree.nodes.get('Principled BSDF')
    if bsdf_trunk:
        bsdf_trunk.inputs['Base Color'].default_value = (0.05, 0.04, 0.03, 1)
        bsdf_trunk.inputs['Roughness'].default_value = 0.9

    trunk.data.materials.append(trunk_mat)

    foliage_mat = bpy.data.materials.new(name="FoliageMaterial")
    foliage_mat.use_nodes = True
    bsdf_foliage = foliage_mat.node_tree.nodes.get('Principled BSDF')
    if bsdf_foliage:
        bsdf_foliage.inputs['Emission Color'].default_value = (0.8, 0.1, 1.0, 1)
        bsdf_foliage.inputs['Emission Strength'].default_value = 10.0
    
    for i in range(15):
        x_offset = random.uniform(-1.5, 1.5)
        y_offset = random.uniform(-1.5, 1.5)
        z_offset = random.uniform(4.5, 6.0)
        
        bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=2, radius=random.uniform(0.3, 0.6), location=(location[0] + x_offset, location[1] + y_offset, location[2] + z_offset))
        leaf = bpy.context.object
        leaf.name = f"Tree_Leaf_{i}"
        leaf.parent = trunk
        leaf.data.materials.append(foliage_mat)
        
    return trunk

def create_futuristic_shrub(location):
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
    parent_empty = bpy.context.object
    parent_empty.name = "Shrub"

    shrub_mat = bpy.data.materials.new(name="ShrubMaterial")
    shrub_mat.use_nodes = True
    nodes = shrub_mat.node_tree.nodes
    bsdf_shrub = nodes.get('Principled BSDF')
    
    tex_coord = nodes.new(type='ShaderNodeTexCoord')
    voronoi = nodes.new(type='ShaderNodeTexVoronoi')
    voronoi.feature = 'DISTANCE_TO_EDGE'
    
    shrub_mat.node_tree.links.new(tex_coord.outputs['Object'], voronoi.inputs['Vector'])
    shrub_mat.node_tree.links.new(voronoi.outputs['Distance'], bsdf_shrub.inputs['Emission Strength'])
    
    if bsdf_shrub:
        bsdf_shrub.inputs['Base Color'].default_value = (0.0, 0.0, 0.0, 1)
        bsdf_shrub.inputs['Emission Color'].default_value = (0.1, 1.0, 0.7, 1)
    
    for i in range(10):
        x = location[0] + random.uniform(-0.8, 0.8)
        y = location[1] + random.uniform(-0.8, 0.8)
        z = location[2] + random.uniform(0, 0.5)
        
        bpy.ops.mesh.primitive_cube_add(size=random.uniform(0.3, 0.7), location=(x,y,z))
        cube = bpy.context.object
        cube.name = f"Shrub_Cube_{i}"
        cube.parent = parent_empty
        cube.data.materials.append(shrub_mat)
        
    return parent_empty
